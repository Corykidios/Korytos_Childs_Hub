function Redact-Secrets($path) {

  $patterns = @(
    'at-let-[A-Za-z0-9]{10,}', 'lin_api_[A-Za-z0-9]{10,}',
    'gho_[A-Za-z0-9]{15,}', 'ghp_[A-Za-z0-9]{20,}',

    'xox[baprs]-[A-Za-z0-9-]{10,}'
  )

  $enc = New-Object System.Text.UTF8Encoding($false)

}
